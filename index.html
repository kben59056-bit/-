دليل تمارين Loops في لغة C

Algorithmique 1 · المستوى الأول الجامعي · أفكار التمارين مع أمثلة وحلول قابلة للإظهار والإخفاء

جميع التمارين ظاهرة: 75 تمريناً في 7 مستويات.
تقدّمك في التمارينأنجزت 0 من 75 تمريناً (0%)
كيف تستعمل الملف؟
1) اقرأ فكرة التمرين وحاول كتابة الحل بنفسك. 2) اضغط «إظهار الحل» للمقارنة. 3) غيّر قيم الإدخال واختبر فهمك.
الحلول مقاطع C تركز على منطق الحلقة. أضف #include <stdio.h> وداخل main عند تحويلها إلى برنامج كامل. في أمثلة الكسور استعملنا double.
ملاحظة: يجب فتح الملف بمتصفح مثل Chrome أو Edge للاستفادة من البحث وتصفية المستويات. حتى إذا لم تعمل JavaScript ستبقى التمارين ظاهرة ويمكن قراءتها.

المستوى 1: أساسيات التكرار · سهل

1. Print numbers

فكرة التمرين

اطبع الأعداد من 1 إلى N. مثال: أدخل N=5، المتوقع: 1 2 3 4 5.

إظهار الحل
int i, n;
scanf("%d", &n);
for(i = 1; i <= n; i++)
    printf("%d ", i);

2. Reverse counting

فكرة التمرين

اطبع الأعداد تنازلياً من N إلى 1. مثال: N=5، المتوقع: 5 4 3 2 1.

إظهار الحل
int i, n;
scanf("%d", &n);
for(i = n; i >= 1; i--)
    printf("%d ", i);

فكرة التمرين

    printf("%d ", i);

فكرة التمرين

المستوى 3: الحلقات مع الشروط (Loops + Conditions) · متوسط

1. Count even and odd

فكرة التمرين

اقرأ N أعداداً واحسب عدد الزوجية والفردية. مثال: لكل عدد اختبر x%2.

إظهار الحل
int i, n, x, even = 0, odd = 0;
scanf("%d", &n);
for(i = 1; i <= n; i++) {
    scanf("%d", &x);
    if(x % 2 == 0) even++;
    else odd++;
}
printf("Even=%d Odd=%d", even, odd);

2. Positive and negative

فكرة التمرين

اقرأ N أعداد واحسب الموجب والسالب والصفر. مثال: استخدم if / else if / else.

إظهار الحل
int i, n, x, pos=0, neg=0, zero=0;
scanf("%d", &n);
for(i=1; i<=n; i++) {
    scanf("%d", &x);
    if(x>0) pos++;
    else if(x<0) neg++;
    else zero++;
}
printf("%d %d %d", pos, neg, zero);

3. Maximum

فكرة التمرين

int i, n, x, max;

scanf("%d", &max);

7. Prime number

فكرة التمرين

تحقق هل N أولي: لا يقبل القسمة إلا على 1 ونفسه. مثال: يكفي الفحص حتى i*i <= n.

إظهار الحل
int i, n, prime = 1;
scanf("%d", &n);
if(n < 2) prime = 0;
for(i=2; i*i<=n && prime; i++)
    if(n % i == 0) prime = 0;
if(prime) printf("Prime");
else printf("Not prime");

8. Prime numbers up to N

فكرة التمرين

اطبع كل الأعداد الأولية من 2 إلى N. مثال: لكل عدد اختبر أوليته.

إظهار الحل
int n, x, d, prime;
scanf("%d", &n);
for(x=2; x<=n; x++) {
    prime = 1;
    for(d=2; d*d<=x; d++)
        if(x%d==0) { prime=0; break; }
    if(prime) printf("%d ", x);
}

9. Perfect number

فكرة التمرين

تحقق هل N يساوي مجموع قواسمه الموجبة الأصغر منه. مثال: 6=1+2+3.

إظهار الحل
int i, n, sum = 0;
scanf("%d", &n);
for(i=1; i<n; i++)
    if(n%i==0) sum += i;
if(n>0 && sum==n) printf("Perfect");
else printf("Not perfect");

10. Divisibility test

فكرة التمرين

اختبر هل N يقبل القسمة على K مع تجنب القسمة على صفر. مثال: الباقي يساوي صفراً.

إظهار الحل
int n, k;
scanf("%d%d", &n, &k);
if(k != 0 && n % k == 0) printf("Divisible");
else printf("Not divisible");

11. GCD / PGCD

فكرة التمرين

احسب القاسم المشترك الأكبر باستعمال خوارزمية إقليدس. مثال: مثال 18 و12 يعطي 6.

إظهار الحل
int a, b, r;
scanf("%d%d", &a, &b);
while(b != 0) {
    r = a % b;
    a = b;
    b = r;
}
printf("%d", a);

12. LCM / PPCM

فكرة التمرين

احسب المضاعف المشترك الأصغر باستعمال PGCD. مثال: LCM(a,b)=|a*b|/GCD.

إظهار الحل
int a, b, x, y, r, gcd;
scanf("%d%d", &a, &b);
x=a; y=b;
while(y!=0) { r=x%y; x=y; y=r; }
gcd=x;
if(a==0 || b==0) printf("0");
else printf("%d", (a/gcd)*b);

13. Input validation

فكرة التمرين

استمر في طلب عدد موجب حتى يدخِل المستخدم قيمة صحيحة. مثال: استعمل do while.

إظهار الحل

do {
    printf("Enter a positive number: ");
    scanf("%d", &n);
} while(n <= 0);

فكرة التمرين

إظهار الحل
